    protected $amadeusService;

    {
        $this->amadeusService = $amadeusService;
    }

    public function index()
    {
        return view('pages.home');
    }

    
    public function search(Request $request)
    {
        //step:1 validate data after user submits the form 
        $request->validate([
            'from_city' => 'required|different:to_city',
            'to_city' => 'required',
            'departureDate' => 'required|date|after_or_equal:today',
            'returnDate' => 'nullable|date|after:departureDate',
            'adults' => 'required|integer|min:1|max:5',
        ]);

        //step2: checks wheather the way input is one way ot two way.....
        $returnDate = $request->input('way') === 'two_way' ? $request->input('returnDate') : null;

        
        //calls the searchFlights method from the AmadeusService class and passes the required parameters 
        $flights = $this->amadeusService->searchFlights(
            $request->input('origin'),
            $request->input('destination'),
            $request->input('departureDate'),
            $returnDate,
            $request->input('adults')
        );

        //It stores the result (the flight offers data) in the $flights variable
        //Error ayo ki nai check garxa ani yo view ma send garxa
        if (isset($flights['error'])) {
            return view('pages.flights.results', ['flights' => $flights, 'searchParams' => $request->all()]);
        }
        $flightDetails = [];
        foreach ($flights as $flight) {
            $segments = $flight->getItineraries()[0]->getSegments();
            $firstSegment = $segments[0];
            $lastSegment = end($segments);


            $currency = $flight->getPrice()->getCurrency();

            $displayCurrency = $currency;
            if ($currency !== 'NPR') {
                $displayCurrency = 'NPR'; // Indicate approximate price
            }

            $duration = $this->calculateTotalDuration($segments);

            $flightDetails[] = [
                'carrierCode' => $firstSegment->getCarrierCode(),
                'flightNumber' => $firstSegment->getNumber(),
                'departureAirport' => $firstSegment->getDeparture()->getIataCode(),
                'arrivalAirport' => $lastSegment->getArrival()->getIataCode(),
                'departureDateTime' => $firstSegment->getDeparture()->getAt(),
                'arrivalDateTime' => $lastSegment->getArrival()->getAt(),
                'travelClass' => $segments[0]->getAircraft()->getCode(),
                'price' => round($flight->getPrice()->getTotal()),
                'currency' => $displayCurrency,
                'duration' =>  $duration,
            ];
        }


        //error xain avani xai yo execute hunxa hai
        
        return view('pages.flightList', ['flightDetails' => $flightDetails, 'searchParams' => $request->all()]);
        // return view('pages.flights.results', ['flightDetails' => $flightDetails, 'searchParams' => $request->all()]);
    }

    private function calculateTotalDuration($segments)
    {
        $totalSeconds = 0;
        foreach ($segments as $segment) {
            $departureTime = new \DateTime($segment->getDeparture()->getAt());
            $arrivalTime = new \DateTime($segment->getArrival()->getAt());
            $interval = $departureTime->diff($arrivalTime);
            $totalSeconds += ($interval->h * 3600) + ($interval->i * 60) + $interval->s;
        }

        $hours = floor($totalSeconds / 3600);
        $minutes = floor(($totalSeconds % 3600) / 60);

        return sprintf('%dh:%02dm', $hours, $minutes);
    }
}

namespace App\Services;

use Amadeus\Amadeus;
use Amadeus\Exceptions\ResponseException;

class AmadeusService
{
    protected $amadeus;

    public function __construct()
    {
        $this->amadeus = Amadeus::builder(
            env('AMADEUS_API_KEY'),
            env('AMADEUS_API_SECRET')
        )->build();
    }

    public function searchFlights($origin, $destination, $departureDate, $returnDate = null, $adults = 1)
    {
        try {
            return $this->amadeus->getShopping()->getFlightOffers()->get([
                'originLocationCode' => $origin,
                'destinationLocationCode' => $destination,
                'departureDate' => $departureDate,
                'returnDate' => $returnDate,
                'adults' => $adults,
                'max' => 5,
                'currencyCode' => 'NPR'
            ]);
        } catch (ResponseException $e) {
            return ['error' => $e->getMessage()];
        }
    }

    public function getAirlineName($carrierCode)
    {
        try {
            $response = $this->amadeus->getReferenceData()->getAirlines()->get([
                'airlineCodes' => $carrierCode
            ]);
    
            if (!empty($response)) {
                return $response[0]->getBusinessName();
            }
        } catch (\Exception $e) {
            // Log or silently fail
        }
    
        return $carrierCode; // fallback
    }
